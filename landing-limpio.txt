const db = firebase.firestore();
const docLanding = db.doc("plataforma/landing");
const colResenas = db.collection("plataforma/landing/resenas");
function esc(s){
  return String(s == null ? "" : s)
    .replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;")
    .replace(/"/g,"&quot;").replace(/'/g,"&#39;");
}
const qs = s => document.querySelector(s);
/* ── Carga del contenido público — corre siempre, sin necesitar la
   clave de administrador. Si el documento no existe todavía (primera
   vez), se queda con el texto que ya está escrito a mano en el HTML,
   sin sobrescribir nada. ── */
async function cargarContenidoPublico(){
  try {
    const doc = await docLanding.get();
    if (!doc.exists) return;
    const d = doc.data();
    if (d.heroTituloPre != null) qs("#heroTituloPre").innerHTML = esc(d.heroTituloPre).replace(/\n/g, "<br>");
    if (d.heroAcento != null) qs("#heroAcento").textContent = d.heroAcento;
    if (d.heroBajada != null) qs("#heroBajada").textContent = d.heroBajada;
    if (d.precioMensual != null) qs("#precioMensual").textContent = d.precioMensual;
    if (d.precioUnico != null) qs("#precioUnico").textContent = d.precioUnico;
    if (d.fraseFooter != null) qs("#fraseFooter").textContent = d.fraseFooter;
  } catch(e){ console.warn("No se pudo cargar el contenido del landing:", e); }
}
async function cargarResenasPublico(){
  try {
    const snap = await colResenas.orderBy("orden").get();
    const grid = qs("#resenasGrid");
    if (snap.empty) return; // se queda con el mensaje "todavía no hay reseñas" que ya está en el HTML
    grid.innerHTML = snap.docs.map(doc => {
      const r = doc.data();
      const iniciales = (r.nombre || "").split(" ").map(p => p[0]).join("").slice(0,2).toUpperCase();
      return `
        <div class="reseña-card">
          <div class="reseña-estrellas">${"★".repeat(Math.max(1,Math.min(5,r.estrellas||5)))}</div>
          <p class="reseña-texto">"${esc(r.texto)}"</p>
          <div class="reseña-autor">
            <div class="reseña-avatar">${esc(iniciales)}</div>
            <div>
              <div class="reseña-nombre">${esc(r.nombre)}</div>
              <div class="reseña-negocio">${esc(r.negocio)}</div>
            </div>
          </div>
        </div>`;
    }).join("");
  } catch(e){ console.warn("No se pudieron cargar las reseñas:", e); }
}
cargarContenidoPublico();
cargarResenasPublico();

   DEVMODE — solo se activa si la URL tiene #dev. Protegido con

   necesita roles.plataforma = "propietario" en usuarios/{uid} — mismo
   mecanismo que ya protege el devmode de cada tienda (ver
   resolverRolPropietario en index.html), reutilizado acá porque el
   devmode del landing escribe directo a Firestore desde el navegador,
   sin pasar por ninguna función de Netlify — así que las reglas de
   Firestore son la única barrera real, y esas reglas solo pueden
   verificar una sesión de Firebase Auth, no un secreto que compara el
   propio JS del cliente.
   ══════════════════════════════════════════════════════════════ */
const auth = firebase.auth();
let ADMIN_SECRET = ""; // se sigue usando aparte, solo para listarSuscripciones.js (ver pestaña Métricas)
function abrirDevGate(){
  qs("#devGate").classList.add("abierto");
}
async function intentarEntrarDev(){
  const email = qs("#devGateEmail").value.trim();
  const pass = qs("#devGateInput").value;
  const err = qs("#devGateErr");
  err.textContent = "";
  if (!email || !pass){ err.textContent = "Completá email y contraseña"; return; }
  try {
    const cred = await auth.signInWithEmailAndPassword(email, pass);
    const doc = await db.doc("usuarios/" + cred.user.uid).get();
    const roles = doc.exists ? (doc.data().roles || {}) : {};
    if (roles.plataforma !== "propietario"){
      await auth.signOut();
      err.textContent = "Esta cuenta no tiene acceso al panel del landing";
      return;
    }
    qs("#devGate").classList.remove("abierto");
    abrirDevPanel();
  } catch(e){
    err.textContent = "Email o contraseña incorrectos";
  }
}
if (location.hash === "#dev"){
  auth.onAuthStateChanged(async user => {
    if (!user){ abrirDevGate(); return; }
    const doc = await db.doc("usuarios/" + user.uid).get();
    const roles = doc.exists ? (doc.data().roles || {}) : {};
    if (roles.plataforma === "propietario") abrirDevPanel();
    else abrirDevGate();
  });
}
qs("#devGateBtn").addEventListener("click", intentarEntrarDev);
qs("#devGateInput").addEventListener("keydown", e => { if (e.key === "Enter") intentarEntrarDev(); });

/* ── Pestaña Contenido — precarga los campos con lo que ya hay
   guardado (o lo que está escrito a mano en el HTML, como valor de
   partida razonable la primera vez). ── */
async function cargarContenidoDev(){
  try {
    const doc = await docLanding.get();
    const d = doc.exists ? doc.data() : {};
    qs("#devHeroTitulo").value = d.heroTituloPre ?? qs("#heroTituloPre").textContent.trim();

    qs("#devPrecioUnico").value = d.precioUnico ?? qs("#precioUnico").textContent.trim();

}

/* ── Pestaña Reseñas — lista editable, cada tarjeta se guarda
   individualmente al perder el foco de cualquiera de sus campos. ── */
async function cargarResenasDev(){
  try {
    const snap = await colResenas.orderBy("orden").get();
    const lista = qs("#devResenasLista");
    lista.innerHTML = snap.docs.map(doc => renderResenaDevCard(doc.id, doc.data())).join("");
    conectarEventosResenas();
  } catch(e){ console.warn(e); }
}
function renderResenaDevCard(id, r){
  return `
    <div class="dev-resena-card" data-id="${esc(id)}">
      <div class="dev-fila">
        <input placeholder="Nombre" value="${esc(r.nombre||"")}" data-campo="nombre">
        <input placeholder="Negocio" value="${esc(r.negocio||"")}" data-campo="negocio">
      </div>
      <div class="dev-fila">
        <input type="number" min="1" max="5" placeholder="Estrellas (1-5)" value="${r.estrellas||5}" data-campo="estrellas" style="max-width:120px">
        <button class="dev-btn-del" data-accion="borrar">Borrar</button>
      </div>
      <textarea placeholder="Texto de la reseña" data-campo="texto">${esc(r.texto||"")}</textarea>
    </div>`;
}
function conectarEventosResenas(){
  document.querySelectorAll(".dev-resena-card").forEach(card => {
    const id = card.dataset.id;
    card.querySelectorAll("[data-campo]").forEach(campo => {
      campo.addEventListener("blur", () => guardarResena(id, card));
    });
    const btnBorrar = card.querySelector("[data-accion='borrar']");
    if (btnBorrar) btnBorrar.addEventListener("click", async () => {
      if (!confirm("¿Borrar esta reseña?")) return;
      await colResenas.doc(id).delete();
      card.remove();
      await cargarResenasPublico();
    });
  });
}
async function guardarResena(id, card){
  try {
    const datos = {
      nombre: card.querySelector('[data-campo="nombre"]').value,
      negocio: card.querySelector('[data-campo="negocio"]').value,
      estrellas: Number(card.querySelector('[data-campo="estrellas"]').value) || 5,
      texto: card.querySelector('[data-campo="texto"]').value
    };
    await colResenas.doc(id).set(datos, { merge: true });
    await cargarResenasPublico();
  } catch(e){ console.warn(e); }
}
qs("#devAgregarResena").addEventListener("click", async () => {
  try {
    const snap = await colResenas.get();
    const nuevoDoc = await colResenas.add({
      nombre: "", negocio: "", estrellas: 5, texto: "", orden: snap.size
    });
    const lista = qs("#devResenasLista");
    lista.insertAdjacentHTML("beforeend", renderResenaDevCard(nuevoDoc.id, { estrellas: 5 }));
    conectarEventosResenas();
  } catch(e){ alert("Error al agregar reseña: " + e.message); }
});

async function cargarMetricasDev(){
  const grid = qs("#devMetricasGrid");
  grid.style.display = "grid";
  grid.innerHTML = '<div class="dev-metrica-card"><div class="num">…</div><div class="lbl">Cargando</div></div>';
  try {
    const res = await fetch("/.netlify/functions/listarSuscripciones", {
      headers: { "X-Admin-Secret": ADMIN_SECRET }
    });
    const j = await res.json().catch(() => ({}));
    if (!res.ok || !j.ok){
      grid.innerHTML = '<p style="color:var(--dev-txt2)">No se pudieron cargar las métricas — revisá la clave.</p>';
      return;
    }
    qs("#devSecretoWrap").style.display = "none";
    const lista = j.suscripciones || [];
    const activas = lista.filter(s => s.estado === "activa").length;
    const totalMensual = lista.filter(s => s.estado === "activa").reduce((sum, s) => sum + (s.montoMensual||0), 0);
    grid.innerHTML = `
      <div class="dev-metrica-card"><div class="num">${lista.length}</div><div class="lbl">Tiendas totales</div></div>
      <div class="dev-metrica-card"><div class="num">${activas}</div><div class="lbl">Suscripciones activas</div></div>
      <div class="dev-metrica-card"><div class="num">$${totalMensual.toLocaleString("es-CL")}</div><div class="lbl">Facturación mensual</div></div>
    `;
  } catch(e){
    grid.innerHTML = '<p style="color:var(--dev-txt2)">Error de red al cargar métricas.</p>';
  }
}
